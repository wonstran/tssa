    Locators¶

    A locator turns rows (box, class and any detection tables) into reference points. The seven built-in locators are registered under their M names and a descriptive alias; your own are added with register_locator.

    Method Alias Needs Scope Classes
    M1 bottom_centre - row (point only) all
    M2 nominal_shift - row all
    M3 nominal_box - row all
    M4 wheel_keypoints keypoints, camera row all
    M5 sam_outline masks row all
    M6 cuboid cuboids row all
    M7 wheel_fit masks, camera segment 5 (bus), 7 (truck)

    locators ¶

    Locator registry: a locator turns rows (bbox, class, extra tables) into native reference points.

    Design docs/2026-11-10-footprint-design.md, section 7.1. A locator is registered with

    @register_locator("M5", needs=("masks",), scope="row", point_only=False)
def locate_m5(rows: pd.DataFrame, ctx: LocateContext) -> LocatorResult: ...
    • scope "row": rows are original rows with a valid projection (fp_row_kind == "box"), any subset.
    • scope "segment": rows are whole segments (consecutive frames, inserted rows included) of one class.
    • classes: the cls_track values the locator serves (None = all); other rows fail with "class_ineligible".
    • A locator whose needs tables were not supplied fails every row with "not_supplied".
    • aliases: further names that resolve to the locator (design 16, Q7), such as "cuboid" for M6. Chains and compare() accept either; results always carry the canonical name (fp_source, fp_fail, method).

    rows carries: track, frame, the box columns (x, y, w, h under their configured names, NaN on inserted rows), cls_track, seg, interp, fp_row_kind, u_bc, v_bc (bbox bottom-centre px), m1_x, m1_y, m2_x, m2_y (world m) and ref_heading_deg.

    def register_locator(name: str, needs=(), scope: str = "row", point_only: bool = False, classes=None, aliases=()):
    """Decorator: register `fn(rows, ctx) -> LocatorResult` as locator `name`, also reachable as each of `aliases`."""
    if scope not in ("row", "segment"):
        raise ValueError(f"scope must be 'row' or 'segment', got {scope!r}")
    if name in ALIASES:
        raise ValueError(f"{name!r} is already an alias of {ALIASES[name]!r}")
    aliases = (aliases,) if isinstance(aliases, str) else tuple(aliases)

    def wrap(fn):
        REGISTRY[name] = Locator(name, fn, tuple(needs), scope, bool(point_only),
                                 None if classes is None else tuple(classes))  # fmt: skip
        for a in aliases:
            register_alias(a, name)
        return fn

    return wrap

    def register_alias(alias: str, name: str) -> None:
    """Make `alias` another name of the registered locator `name`."""
    if name not in REGISTRY:
        raise ValueError(f"unknown locator {name!r}; known: {sorted(REGISTRY)}")
    if alias in REGISTRY:
        raise ValueError(f"{alias!r} is a locator name, not available as an alias")
    if ALIASES.get(alias, name) != name:
        raise ValueError(f"{alias!r} is already an alias of {ALIASES[alias]!r}")
    ALIASES[alias] = name

    def canonical(name: str) -> str:
    """The registered name behind `name` (itself, or the locator an alias points to); ValueError if unknown."""
    if name in REGISTRY:
        return name
    if name in ALIASES and ALIASES[name] in REGISTRY:
        return ALIASES[name]
    raise ValueError(f"unknown locator {name!r}; known: {sorted(REGISTRY)}, aliases: {sorted(ALIASES)}")

    def solve_m3(ground, box_xywh, heading_deg, L_m, W_m, init, resid_ok_px: float = 0.5):
    """(centre, corners, ok, resid_px): the nominal rectangle at `heading_deg` whose projection has its lowest point
    on the bbox bottom and its horizontal midpoint at the bbox centre (distorted pixels, like the bbox)."""
    nan2, nan42 = np.full(2, np.nan), np.full((4, 2), np.nan)
    init = np.asarray(init, dtype=float)
    if not np.isfinite(heading_deg) or not np.isfinite(init).all():
        return nan2, nan42, False, np.nan
    x, y, w, h = (float(v) for v in box_xywh)
    tv, tu = y + h, x + w / 2.0

    def resid(d):
        uv = np.asarray(ground.world_to_image(rect_corners(init + d, heading_deg, L_m, W_m)), dtype=float)
        return np.array([uv[:, 1].max() - tv, (uv[:, 0].min() + uv[:, 0].max()) / 2.0 - tu])

    try:
        r0 = resid(np.zeros(2))
        if not np.isfinite(r0).all():
            return nan2, nan42, False, np.nan
        sol = least_squares(resid, np.zeros(2), method="lm", x_scale=1.0, xtol=1e-12, ftol=1e-12, gtol=1e-12,
                            max_nfev=200)  # fmt: skip
    except (ValueError, FloatingPointError):
        return nan2, nan42, False, np.nan
    r = float(np.max(np.abs(sol.fun)))
    if not np.isfinite(r):
        return nan2, nan42, False, np.nan
    centre = init + sol.x
    return centre, rect_corners(centre, heading_deg, L_m, W_m), bool(r < resid_ok_px), r

    @register_locator("M3", aliases="nominal_box")
def locate_m3(rows, ctx) -> LocatorResult:
    """M3 at the reference heading; the orientation is imposed (directed psi_ref), not measured."""
    cfg = ctx.config
    n = len(rows)
    res = LocatorResult.empty(n)
    boxes = rows[list(cfg.box_cols)].to_numpy(dtype=float)
    psi = rows["ref_heading_deg"].to_numpy(dtype=float)
    init = rows[["m2_x", "m2_y"]].to_numpy(dtype=float)
    cls = rows["cls_track"].to_numpy()
    resid = np.full(n, np.nan)
    for i in range(n):
        if not np.isfinite(psi[i]):
            res.fail_reason[i] = "no_heading"
            continue
        L, W = (v * FT for v in cfg.size_of(cls[i]))
        c, _, ok, r = solve_m3(ctx.ground, boxes[i], psi[i], L, W, init[i], cfg.m3_resid_px)
        resid[i] = r
        if ok:
            res.ok[i], res.x[i], res.y[i] = True, c[0], c[1]
            res.orient_deg[i], res.orient_kind[i] = psi[i], "directed"
        else:
            res.fail_reason[i] = "no_solution"
    res.extras["m3_resid_px"] = resid
    return res

    m4 ¶

    M4: CarFusion wheel keypoints -> body centre and directed heading (ported from the speed study's m4.py; base design section 6.1).

    Keypoint schema: 0 right-front, 1 left-front, 2 right-back, 3 left-back; only the same-side pairs right (0, 2) and left (1, 3) are used. Keypoints are distorted pixels (like the bbox); to_ground maps them to world contacts (ground.image_to_world for contact landmarks, hub_to_contact through the camera for hub landmarks, None when hub landmarks meet a camera that failed its consistency check: camera_invalid). Image heights ("lower in the image", the bbox-bottom cue) use the ground contacts re-projected with world_to_image, in the same pixels as the bbox.

    def solve_m4(kp_uv, box_xywh, psi_ref_deg, cls, dh, to_ground, cfg: FootprintConfig | None = None) -> M4Result:
    """Body centre (world) and directed heading from four wheel keypoints [RF, LF, RB, LB] (distorted px)."""
    cfg = _cfg(cfg)
    if to_ground is None:
        return M4Result(fail_reason="camera_invalid")
    if int(cls) not in cfg.priors_ft:
        return M4Result(fail_reason="no_prior")
    kp_uv = np.asarray(kp_uv, dtype=float).reshape(4, 2)
    psi = float(psi_ref_deg)
    pri = cfg.priors_ft[int(cls)]
    wb_m, tr_m = pri["wheelbase"] * FT, pri["track"] * FT
    x, y, w, h = (float(v) for v in box_xywh)

    G = np.full((4, 2), np.nan)
    fin = np.isfinite(kp_uv).all(axis=1)
    if fin.any():
        G[fin] = np.asarray(to_ground(kp_uv[fin]), dtype=float).reshape(-1, 2)
    gfin = np.isfinite(G).all(axis=1)
    v_img = np.full(4, np.nan)  # image v of the ground contacts (same pixels as the bbox)
    if gfin.any():
        v_img[gfin] = np.asarray(dh.world_to_image(G[gfin]), dtype=float).reshape(-1, 2)[:, 1]

    # 1. Acceptance; the lower passing pair (larger mean v) is chosen.
    passing = [(side, np.mean(v_img[list(ij)])) for side, ij in PAIRS.items()
               if np.isfinite(psi) and _pair_accept(G[ij[0]], G[ij[1]], psi, wb_m, cfg)]  # fmt: skip
    if not passing:
        return M4Result(fail_reason="no_valid_pair")
    side = max(passing, key=lambda t: t[1] if np.isfinite(t[1]) else -np.inf)[0]
    other = "L" if side == "R" else "R"
    (fi, bi), (ofi, obi) = PAIRS[side], PAIRS[other]
    F, B = G[fi], G[bi]
    res = M4Result(pair_side=side, wheelbase_ft=float(np.linalg.norm(F - B) / FT))

    # 2. Front.
    d_front = (F - B) / np.linalg.norm(F - B)
    res.front_flipped = bool(diff_directed(compass_deg(d_front), psi) > 90.0)
    if res.front_flipped:
        d_front = -d_front
    res.d_front = d_front
    n_left = rot_ccw(d_front)
    axle = (F + B) / 2.0

    # 3. Interior side: credible other pair, else camera-away, else ambiguous.
    reject, s = _credibility(G[ofi], G[obi], kp_uv[[ofi, obi]], axle, d_front, n_left, psi, wb_m, tr_m, cfg)
    res.side_reject = reject
    sign = 0.0
    if reject == "":
        sign = float(np.sign(s))
        res.interior_source = "other_pair"
        res.track_ft = abs(s) / FT
    else:
        v_ch, v_ot = v_img[[fi, bi]], v_img[[ofi, obi]]
        depth_ok = (not np.isfinite(v_ot).all()) or bool(v_ch.mean() >= v_ot.mean() + cfg.m4_near_dv_px)
        bottom_ok = bool(np.any(np.abs((y + h) - v_ch) <= cfg.m4_near_bottom_frac * h))  # two-sided (R8)
        if depth_ok and bottom_ok:
            a = away_dir(dh, np.asarray(dh.world_to_image(axle[None, :]), dtype=float))[0]
            dot = float(a @ n_left)
            if np.isfinite(dot) and abs(dot) >= cfg.m4_camera_away_dot:
                sign = float(np.sign(dot))
                res.interior_source = "camera_away"
    if sign == 0.0:
        res.fail_reason = "side_ambiguous"
        return res

    # 4. Centre. 5. Label case: interior +n_left means the pair is physically the vehicle's right side.
    res.centre = axle + (tr_m / 2.0) * sign * n_left + ((pri["front_oh"] - pri["rear_oh"]) * FT / 2.0) * d_front
    geo_side = "R" if sign > 0 else "L"
    res.label_case = _LABEL_CASE[(res.front_flipped, geo_side != side)]
    res.valid = True
    return res

    def to_ground_for(ctx, cls):

    def front_conflict(track, valid, flipped, min_rows: int, share: float) -> np.ndarray:
    """Per row: its track has at least `min_rows` native-valid rows and more than `share` of them front-flipped."""
    df = pd.DataFrame({"track": track, "ok": valid, "f": flipped & valid})
    g = df[df["ok"]].groupby("track")["f"].agg(["size", "mean"])
    flag = (g["size"] >= min_rows) & (g["mean"] > share)
    return df["track"].map(flag).fillna(False).to_numpy(dtype=bool)

    @register_locator("M4", needs=("keypoints",), aliases="wheel_keypoints")
def locate_m4(rows, ctx) -> LocatorResult:
    """Fail reasons: no_match, no_heading, camera_invalid, no_valid_pair, side_ambiguous, no_prior."""
    cfg = ctx.config
    n = len(rows)
    res = LocatorResult.empty(n, "no_match")
    kp = ctx.tables["keypoints"]
    missing = [c for c in ["track", "frame", *KP_COLUMNS] if c not in kp.columns]
    if missing:
        raise ValueError(f"keypoints table lacks column(s) {missing}")
    hit = kp.drop_duplicates(["track", "frame"]).set_index(["track", "frame"])
    hit = hit.reindex(list(zip(rows["track"].to_numpy(), rows["frame"].to_numpy())))
    kp_arr = hit[KP_COLUMNS].to_numpy(dtype=float)
    present = hit[KP_COLUMNS].notna().any(axis=1).to_numpy()
    boxes = rows[list(cfg.box_cols)].to_numpy(dtype=float)
    psi = rows["ref_heading_deg"].to_numpy(dtype=float)
    cls = rows["cls_track"].to_numpy()
    diag = {c: np.full(n, "", dtype=object) for c in ("m4_pair_side", "m4_label_case", "m4_interior_source",
                                                       "m4_side_reject")}  # fmt: skip
    flipped = np.zeros(n, bool)
    wb, tr = np.full(n, np.nan), np.full(n, np.nan)
    grounds: dict = {}
    for i in range(n):
        if not present[i]:
            continue
        if not np.isfinite(psi[i]):
            res.fail_reason[i] = "no_heading"
            continue
        c = int(cls[i])
        if c not in grounds:
            grounds[c] = to_ground_for(ctx, c)
        r = solve_m4(kp_arr[i].reshape(4, 2), boxes[i], psi[i], c, ctx.ground, grounds[c], cfg)
        res.fail_reason[i] = r.fail_reason
        diag["m4_pair_side"][i], diag["m4_label_case"][i] = r.pair_side, r.label_case
        diag["m4_interior_source"][i], diag["m4_side_reject"][i] = r.interior_source, r.side_reject
        flipped[i], wb[i], tr[i] = r.front_flipped, r.wheelbase_ft, r.track_ft
        if r.valid:
            res.ok[i] = True

    def mirror_away(pts, centre, axis_deg, L_m, W_m, away) -> tuple[np.ndarray, bool]:
    """(centre, flipped): unchanged when (centre - outline mean) . away > 0, else reflected across the rectangle side
    closest to the outline points (R20)."""
    pts = np.asarray(pts, dtype=float)
    c = np.asarray(centre, dtype=float)
    if float((c - pts.mean(axis=0)) @ np.asarray(away, dtype=float)) > 0:
        return c, False
    th = np.radians(axis_deg)
    t = np.array([np.sin(th), np.cos(th)])
    s = np.array([-t[1], t[0]])
    best, offset = np.inf, None
    for normal, h, along, ha in ((s, W_m / 2.0, t, L_m / 2.0), (t, L_m / 2.0, s, W_m / 2.0)):
        for sign in (1.0, -1.0):
            off = sign * h * normal  # centre -> side midpoint
            d = pts - (c + off)
            a = np.clip(d @ along, -ha, ha)
            dist = float(np.mean(np.linalg.norm(d - a[:, None] * along, axis=1)))
            if dist < best:
                best, offset = dist, off
    return c + 2.0 * offset, True

    def solve_m5(outline_px, box_xywh, ground, L_m, W_m, init, min_cols: int = 8, max_resid_ft: float = 1.0) -> dict:
    """One row: outline pixels -> dict(ok, reason, centre, axis, cols, resid_ft, sides, flipped)."""
    out = dict(ok=False, reason="", centre=np.full(2, np.nan), axis=np.nan, cols=0, resid_ft=np.nan, sides=0,
               flipped=False)  # fmt: skip
    if outline_px is None:
        out["reason"] = "no_match"
        return out
    x, y, w, h = (float(v) for v in box_xywh)
    o = np.asarray(outline_px, dtype=float).reshape(-1, 2)
    o = o[~(o[:, 1] > y + h + BELOW_BOTTOM_PX)]  # R23
    out["cols"] = len(o)
    if len(o) < min_cols:
        out["reason"] = "few_columns"
        return out
    pts = project(ground, o)
    try:
        c, ax, r, s = fit_rect(pts, L_m, W_m, init)
        c, flipped = mirror_away(pts, c, ax, L_m, W_m, away_dir(ground, o.mean(axis=0))[0])
        if flipped:
            r, s = _resid_sides(pts, c, np.radians(ax), L_m, W_m)
    except (ValueError, FloatingPointError):
        out["reason"] = "fit_residual"
        return out
    out.update(resid_ft=r, sides=s, flipped=flipped)
    if not np.isfinite(r) or r > max_resid_ft:
        out["reason"] = "fit_residual"
        return out
    out.update(ok=True, centre=c, axis=ax)
    return out

def locate_m5(rows, ctx) -> LocatorResult:
    """Fail reasons: no_match (no mask), few_columns, fit_residual."""
    cfg = ctx.config
    n = len(rows)
    res = LocatorResult.empty(n)
    masks = ctx.tables["masks"]
    boxes = rows[list(cfg.box_cols)].to_numpy(dtype=float)
    init = rows[["m2_x", "m2_y"]].to_numpy(dtype=float)
    cls = rows["cls_track"].to_numpy()

        polys = masks.get(key) if hasattr(masks, "get") else None
        outline = raw if raw is not None else (lower_outline(list(polys), boxes[i]) if polys else None)
        L, W = (v * FT for v in cfg.size_of(cls[i]))

            res.fail_reason[i] = r["reason"]

    return res

    @register_locator("M6", needs=("cuboids",), aliases="cuboid")
def locate_m6(rows, ctx) -> LocatorResult:
    """Fail reasons: no_match (no row, IoU below m6_min_iou, non-finite corners); implausible_size (length or width
    below m6_size_gate of the class's nominal size: a collapsed face)."""
    cfg = ctx.config
    n = len(rows)
    res = LocatorResult.empty(n, "no_match")
    cub = ctx.tables["cuboids"]
    missing = [c for c in ["track", "frame", *UV_COLS] if c not in cub.columns]
    if missing:
        raise ValueError(f"cuboids table lacks column(s) {missing}")
    cub = cub.drop_duplicates(["track", "frame"]).set_index(["track", "frame"])
    hit = cub.reindex(list(zip(rows["track"].to_numpy(), rows["frame"].to_numpy())))
    uv = hit[UV_COLS].to_numpy(dtype=float)
    iou = hit["iou"].to_numpy(dtype=float) if "iou" in hit.columns else np.where(np.isfinite(uv).all(1), 1.0, np.nan)
    cls = rows["cls_track"].to_numpy()
    raw = np.full((n, 8), np.nan)
    for i in range(n):
        if not (iou[i] >= cfg.m6_min_iou) or not np.isfinite(uv[i]).all():
            continue
        corners, cen, L, W, axis = face_geometry(uv[i].reshape(4, 2), ctx.ground)
        nom_l, nom_w = cfg.size_of(cls[i])
        if L < cfg.m6_size_gate * nom_l * FT or W < cfg.m6_size_gate * nom_w * FT:
            res.fail_reason[i] = "implausible_size"
            continue
        res.ok[i], res.fail_reason[i] = True, ""
        res.x[i], res.y[i] = cen
        res.orient_deg[i], res.orient_kind[i] = axis, "axis"
        res.len_ft[i], res.wid_ft[i] = L / FT, W / FT
        raw[i] = corners.ravel()
    res.extras.update({c: raw[:, j] for j, c in enumerate(RAW_COLS)})
    res.extras["m6_len_ft"], res.extras["m6_wid_ft"] = res.len_ft.copy(), res.wid_ft.copy()
    res.extras["m6_iou"] = np.where(res.ok, iou, np.nan)
    return res
