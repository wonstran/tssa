    Masks¶

    masks ¶

    Mask contours per (track, frame): the masks table of M5 and M7, and the outline helpers (ported from the speed study's m5.lower_outline and m7_candidates.split_contours / join_contours / rasterise).

    A MaskTable maps (track, frame) -> list of external contour polygons (k, 2) in distorted pixels, largest first. On disk it is the study's m7 cache layout: an .npz with arrays track, frame, count and pts, where pts holds every row's contours back to back, each polygon followed by a NaN row, and count[i] rows of pts belong to row i. Rasterising needs OpenCV (imported on first use).

    @classmethod
def load_npz(cls, path) -> "MaskTable":
    """Read the layout `save_npz` writes (also the speed study's m7 cache `outlines.npz`)."""
    z = np.load(path, allow_pickle=False)
    track, frame, count, pts = z["track"], z["frame"], z["count"], z["pts"]  # each array read once
    ends = np.cumsum(count)
    starts = ends - count
    data = {(t, f): split_contours(pts[a:b]) for t, f, a, b in zip(track, frame, starts, ends)}
    outl = {}
    if "o_pts" in z.files:
        ot, of, oc, op = z["o_track"], z["o_frame"], z["o_count"], z["o_pts"]
        oe = np.cumsum(oc)
        outl = {(t, f): op[a:b] for t, f, a, b in zip(ot, of, oe - oc, oe)}
    return cls(data, outl)

    def split_contours(arr) -> list[np.ndarray]:
    """(n, 2) array with NaN separator rows -> list of polygons."""
    arr = np.asarray(arr, dtype=float).reshape(-1, 2)
    if not len(arr):
        return []
    sep = np.isnan(arr).any(axis=1)
    out, start = [], 0
    for k in np.flatnonzero(sep):
        if k > start:
            out.append(arr[start:k].copy())
        start = k + 1
    if start < len(arr):
        out.append(arr[start:].copy())
    return out

    def join_contours(polys) -> np.ndarray:
    """List of polygons -> (n, 2) with a NaN row after each polygon."""
    parts = []
    for p in polys:
        parts.append(np.asarray(p, dtype=float).reshape(-1, 2))
        parts.append(np.full((1, 2), np.nan))
    return np.vstack(parts) if parts else np.empty((0, 2))

    def rasterise(polys, img_size) -> np.ndarray:
    """Bool (H, W) mask of the filled polygons."""
    cv2 = _cv2()
    W, H = (int(round(v)) for v in img_size)
    canvas = np.zeros((H, W), np.uint8)
    pts = [np.round(np.asarray(p)).astype(np.int32) for p in polys if len(p) >= 3]
    if pts:
        cv2.fillPoly(canvas, pts, 1)
    return canvas.astype(bool)

    def component_areas(polys) -> np.ndarray:
    """Area + perimeter samples of each contour polygon (the study's `cv2.contourArea(c) + len(c)`, by the
    shoelace formula, so thin parts still count)."""
    out = []
    for p in polys:
        p = np.asarray(p, dtype=float).reshape(-1, 2)
        x, y = p[:, 0], p[:, 1]
        out.append(0.5 * abs(float(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))) + len(p))
    return np.asarray(out, dtype=float)

    def keep_components(polys, frac: float) -> list:
    """Components holding at least `frac` of the summed area, largest first (study infer.mask_contours); applying it
    again with the same `frac` keeps them all."""
    polys = list(polys)
    if not polys:
        return []
    a = component_areas(polys)
    return [polys[i] for i in np.argsort(-a) if a[i] >= frac * a.sum()]

    def lower_outline(mask, box_xywh) -> np.ndarray:
    """(m, 2) pixels [u, v]: the largest set row per integer column inside the bbox.

    `mask` is a polygon (k, 2) of pixel vertices, a list of polygons, or an (H, W) bool array.
    """
    x, y, w, h = (float(v) for v in box_xywh)
    if isinstance(mask, (list, tuple)):
        polys = [np.asarray(p, dtype=float).reshape(-1, 2) for p in mask]
        if not polys:
            return np.empty((0, 2))
        hi = np.vstack(polys).max(axis=0)
        m = rasterise(polys, (max(hi[0], x + w) + 2, max(hi[1], y + h) + 2))
    else:
        m = np.asarray(mask)
        if m.ndim == 2 and m.shape[1] == 2 and m.dtype != bool:
            if len(m) < 3:
                return np.empty((0, 2))
            m = rasterise([m], (int(np.ceil(max(m[:, 0].max(), x + w))) + 2, int(np.ceil(max(m[:, 1].max(), y + h))) + 2))
    H, Wd = m.shape
    u0, u1 = max(int(np.ceil(x)), 0), min(int(np.floor(x + w)), Wd - 1)
    out = []
    for u in range(u0, u1 + 1):
        rows = np.flatnonzero(m[:, u])
        if len(rows):
            out.append((float(u), float(rows[-1])))
    return np.array(out, dtype=float).reshape(-1, 2)
