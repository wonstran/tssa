    Detection¶

    The detectors need the optional install pip install tssa[detect]. run() reads the video once and runs every detector on the frames it has rows for. Its output tables go straight into VehFootPrinter.estimate(tracks, **tables).

    runner ¶

    Cached detection runner (design docs/2026-11-10-footprint-design.md section 6.4; study infer.py _run).

    run(video, tracks, detectors, rows=None, cache_dir=None): - tracks is the MATCHING CONTEXT: every box of every frame, always used whole; - rows selects the OUTPUT (and cached) rows: None = every row of tracks for every detector; a DataFrame for all detectors; or {table name: DataFrame} (e.g. VehFootPrinter.required_rows(tracks)); - frames are read once, in order, and every detector with rows in a frame runs on it; - cache (optional): <cache_dir>/<table>/<key>/ with rows.csv, meta.json and (masks) outlines.npz; the key hashes the video file, the full track table, the weights, the detector parameters and its code version. A changed input gives a new key, never silent reuse; a run infers only rows the cache lacks and saves every SAVE_EVERY frames. video is a path (read with OpenCV) or a mapping {frame: BGR image} (in-memory use and tests).

    def join_contours(polys) -> np.ndarray:
    """List of polygons -> (n, 2) with a NaN row after each polygon."""
    parts = []
    for p in polys:
        parts.append(np.asarray(p, dtype=float).reshape(-1, 2))
        parts.append(np.full((1, 2), np.nan))
    return np.vstack(parts) if parts else np.empty((0, 2))

    def split_contours(arr) -> list[np.ndarray]:
    """(n, 2) array with NaN separator rows -> list of polygons."""
    arr = np.asarray(arr, dtype=float).reshape(-1, 2)
    if not len(arr):
        return []
    sep = np.isnan(arr).any(axis=1)
    out, start = [], 0
    for k in np.flatnonzero(sep):
        if k > start:
            out.append(arr[start:k].copy())
        start = k + 1
    if start < len(arr):
        out.append(arr[start:].copy())
    return out

    def table_sha(df: pd.DataFrame) -> str:
    """sha256 of the canonical CSV of track, frame, x, y, w, h (sorted)."""
    t = _norm(df).sort_values(["track", "frame"]).reset_index(drop=True)

    def iter_frames(video, frames):
    """Yield (frame, image) for the requested zero-based frames, in order. A mapping is read directly; a path is
    read sequentially with OpenCV, and a frame that cannot be read raises (frames never shift)."""
    wanted = sorted({int(f) for f in frames})
    if wanted and wanted[0] < 0:
        raise ValueError(f"negative frame number {wanted[0]}")
    if isinstance(video, Mapping):
        for f in wanted:
            if f not in video:
                raise RuntimeError(f"frame {f} is not in the supplied images")
            yield f, video[f]
        return
    from .detectors import _need

    cv2 = _need("cv2")
    cap = cv2.VideoCapture(str(video))
    if not cap.isOpened():
        raise RuntimeError(f"video could not be opened: {video}")
    pos = 0
    try:
        for f in wanted:
            while pos < f:
                if not cap.grab():
                    raise RuntimeError(f"frame {f} could not be read (the video stops at frame {pos})")
                pos += 1
            ok, img = cap.read()
            if not ok or img is None:
                raise RuntimeError(f"frame {f} could not be read")
            pos += 1
            yield f, img
    finally:
        cap.release()

    def run(video, tracks: pd.DataFrame, detectors, rows=None, cache_dir=None, models: dict | None = None) -> dict:
    """{table: result} for each detector: keypoints and cuboids as DataFrames of matched rows, masks as a MaskTable.

    `models`: optional {table: loaded model}, to skip loading. See the module docstring for the rest.
    """
    ctx_all = _norm(tracks)
    tracks_id = table_sha(ctx_all)
    video_id = file_sha(video) if not isinstance(video, Mapping) else "in-memory"
    entries = []
    for det in detectors:
        want = rows.get(det.table) if isinstance(rows, Mapping) else rows
        want = ctx_all if want is None else _norm(want)
        e = _Entry(det, video_id, tracks_id, cache_dir)
        e.want = want
        e.todo = want[~pd.MultiIndex.from_frame(want[["track", "frame"]]).isin(e.have())]
        entries.append(e)
    frames = sorted(set().union(*[set(e.todo["frame"].unique()) for e in entries])) if entries else []
    if frames:
        loaded = dict(models or {})
        by = {id(e): {f: g for f, g in e.todo.groupby("frame")} for e in entries}
        for e in entries:
            if len(e.todo) and e.det.table not in loaded:
                loaded[e.det.table] = e.det.load()
        ctx_by = {f: g.sort_values("track", kind="stable").reset_index(drop=True)
                  for f, g in ctx_all[ctx_all["frame"].isin(frames)].groupby("frame")}  # fmt: skip
        t0, n = time.perf_counter(), 0
        try:
            for f, img in iter_frames(video, frames):
                for e in entries:
                    req = by[id(e)].get(f)
                    if req is None:
                        continue
                    ctx = pd.concat([ctx_by.get(f, req.iloc[:0]), req]).drop_duplicates(["track", "frame"])
                    ctx = ctx.sort_values("track", kind="stable").reset_index(drop=True)
                    recs, outs = e.det.infer(loaded[e.det.table], img, req, ctx)
                    e.pending.extend(recs)
                    e.pending_out.update(outs)
                n += 1
                if n % SAVE_EVERY == 0:
                    for e in entries:
                        e.flush(time.perf_counter() - t0)
                    log.info("detect: %d of %d frames", n, len(frames))
        finally:
            for e in entries:
                e.flush(time.perf_counter() - t0)
    out = {}
    for e in entries:
        key = pd.MultiIndex.from_frame(e.want[["track", "frame"]])
        got = e.df[pd.MultiIndex.from_frame(e.df[["track", "frame"]]).isin(key)] if len(e.df) else e.df
        got = got.sort_values(["track", "frame", "iou"], ascending=[True, True, False])
        got = got.drop_duplicates(["track", "frame"]).reset_index(drop=True)
        if e.det.table == "masks":
            keys = set(map(tuple, got.loc[got["matched"].astype(bool), ["track", "frame"]].to_numpy()))
            out["masks"] = MaskTable({k: v for k, v in e.outlines.items() if k in keys},
                                     {k: v for k, v in e.lower.items() if k in keys})
        else:
            m = got[got["matched"].astype(bool)][e.det.columns].reset_index(drop=True)
            m[["track", "frame"]] = m[["track", "frame"]].astype("int64")
            out[e.det.table] = m
        if e.dir is not None:
            log.info("detect: %s cache %s", e.det.table, e.dir)
    return out

    detectors ¶

    Model wrappers (optional install tssa[detect]): keypoints, 3-D cuboids, SAM masks.

    Ported from the speed study's infer.py (snapshot docs/study-snapshot-2026-11-10.json). Every model import is lazy.     def mask_contours(m: np.ndarray, min_frac: float) -> list:
    """External contours (k, 2) of a bool mask, components with >= min_frac of its area, largest first."""
    cv2 = _need("cv2")
    cs, _ = cv2.findContours(m.astype(np.uint8), cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
    if not cs:
        return []
    areas = np.array([cv2.contourArea(c) + len(c) for c in cs], dtype=float)  # + perimeter: thin parts count
    keep = [i for i in np.argsort(-areas) if areas[i] >= min_frac * areas.sum()]
    return [cs[i].reshape(-1, 2).astype(float) for i in keep]

    def compute_iou(box1, boxes2) -> np.ndarray:
    """IoU between a box [x1, y1, x2, y2] and an (N, 4) array of boxes."""
    boxes2 = np.asarray(boxes2, dtype=float).reshape(-1, 4)
    if len(boxes2) == 0:
        return np.zeros((0,), dtype=float)
    box1 = np.asarray(box1, dtype=float)
    ix1 = np.maximum(box1[0], boxes2[:, 0])
    iy1 = np.maximum(box1[1], boxes2[:, 1])
    ix2 = np.minimum(box1[2], boxes2[:, 2])
    iy2 = np.minimum(box1[3], boxes2[:, 3])
    inter = np.maximum(0.0, ix2 - ix1) * np.maximum(0.0, iy2 - iy1)
    area1 = max(0.0, box1[2] - box1[0]) * max(0.0, box1[3] - box1[1])
    area2 = np.maximum(0.0, boxes2[:, 2] - boxes2[:, 0]) * np.maximum(0.0, boxes2[:, 3] - boxes2[:, 1])
    return inter / np.maximum(area1 + area2 - inter, 1e-6)

    def match_frame(track_xyxy, det_xyxy, min_iou: float = 0.3, det_cls=None, prefer=None) -> tuple[np.ndarray, np.ndarray]:
    """One-to-one detection -> track-box assignment for one frame: (detection index or -1, IoU) per track box.

    Pairs below `min_iou` are excluded and the rest are assigned by maximum total IoU (linear_sum_assignment on
    -IoU); a pair is kept only if its IoU >= min_iou (and > 0). With `det_cls` and `prefer`, only detections of a
    preferred class are candidates whenever the frame has any. The IoU of an unmatched box is its best candidate IoU
    (diagnostic). Pass every track box of the frame, in a fixed order, so a row's assignment does not depend on
    which rows were requested.
    """
    from scipy.optimize import linear_sum_assignment

    t = np.asarray(track_xyxy, dtype=float).reshape(-1, 4)
    dets = np.asarray(det_xyxy, dtype=float).reshape(-1, 4)
    idx, best = np.full(len(t), -1, dtype=int), np.zeros(len(t))
    if len(t) == 0 or len(dets) == 0:
        return idx, best
    cand = np.ones(len(dets), dtype=bool)
    if det_cls is not None and prefer is not None:
        ok = np.isin(np.asarray(det_cls).astype(int), list(prefer))
        if ok.any():
            cand = ok
    M = np.vstack([compute_iou(b, dets) for b in t])
    M[:, ~cand] = 0.0
    best = M.max(axis=1)
    A = np.where(M >= min_iou, M, 0.0)
    for i, j in zip(*linear_sum_assignment(-A)):
        if A[i, j] > 0.0:
            idx[i], best[i] = j, A[i, j]
    return idx, best
