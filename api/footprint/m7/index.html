    M7 Rigid Fit¶

    M7 fits a rigid body with two axles to a whole bus or truck segment, anchored on the tyres found in the SAM masks. It is the slowest method, about a minute per segment on one core. VehFootPrinter(workers=..., cache_dir=...) runs the segments in parallel and caches each solve.

    m7 ¶

    M7: wheel-anchored rigid-body fit per bus or truck segment (M7 design rev 7 plus the M7 notes' decisions to 2026-11-10; footprint design section 7.6). Ported from the speed study's m7.py, m7_candidates.py and the job, cache and table parts of m7_runner.py (snapshot docs/study-snapshot-2026-11-10.json).

    Per segment of a cls 5/7 track: frozen observations from the mask contours (frame_obs_from_mask, the other boxes of each frame from the full track table), an initial pose from the M2 points and psi_ref, solve_track, then rows: wheels2 / wheel_front / wheel_rear / silhouette rows are native; kinematic rows carry the solve's pose and centre covariance for admission (design 7.7); a failed solve fails every row with its gate reason. Solves run in workers spawned processes (one BLAS thread each) and are cached under <cache_dir>/m7solve/.

    def init_pose(m2_local: np.ndarray, psi: np.ndarray):
    """Initial centre (M2 points interpolated across rows without one) and heading (psi_ref interpolated; else the
    path direction), study m7_runner._init_pose. None when no row has an M2 point."""
    ok = np.isfinite(m2_local).all(axis=1)
    if not ok.any():
        return None
    c = m2_local.copy()
    c[~ok] = np.nan
    c0 = pd.DataFrame(c).interpolate(limit_direction="both").to_numpy(float)
    p = _interp_angle(psi)
    if not np.isfinite(p).all():
        d = np.gradient(c0, axis=0) if len(c0) > 1 else np.array([[0.0, 1.0]])
        p = np.degrees(np.arctan2(d[:, 0], d[:, 1])) % 360
    return c0, p

    def job_key(tr, ns: dict, camera_ok: bool, ident) -> str:
    """Hash of the solve's inputs: code, settings, identity, frames, initial pose, camera and frozen observations."""
    h = hashlib.sha256()
    h.update(code_version().encode())
    h.update(json.dumps(ns, sort_keys=True, default=str).encode())
    h.update(json.dumps([*ident, bool(camera_ok)], default=str).encode())
    _upd(h, tr.frames)
    for a in (tr.origin, tr.init_centre, tr.init_psi_deg, [tr.cls, tr.dt]):
        _upd(h, np.asarray(a, float))
    for k, v in sorted(vars(tr.cam).items()):
        h.update(k.encode())
        _upd(h, np.asarray(v, float)) if np.asarray(v).dtype.kind in "biuf" else h.update(repr(v).encode())
    for o in tr.obs:
        _upd(h, [o.frame])
        for a in (o.cand, o.kappa, o.env_u, o.env_v, o.ext, o.cand_s):
            _upd(h, np.asarray(a, float))
    return h.hexdigest()

    def solve_one(tr, ns: dict, camera_ok: bool = True) -> dict:
    """One solve -> {"frames", "summary", "seconds"} or {"error", "traceback", "seconds"}. Never raises."""
    t0 = time.perf_counter()
    try:
        with using(ns):
            res = m7.solve_track(tr, camera_ok=camera_ok)
        out = {"frames": res["frames"], "summary": res["summary"]}
    except Exception as e:  # noqa: BLE001 - one bad segment must not lose the run; it is reported
        out = {"error": f"{type(e).__name__}: {e}", "traceback": traceback.format_exc()}
    out["seconds"] = time.perf_counter() - t0
    return out

    def solve_jobs(jobs: list[dict], ns: dict, cache_dir=None, workers: int = 1, camera_ok: bool = True) -> dict:
    """{key: result}: cached results reused; the rest solved longest first, in `workers` spawned processes."""
    out, todo = {}, []
    for j in jobs:
        p = _cache_path(cache_dir, j["key"]) if cache_dir is not None else None
        if p is not None and p.is_file():
            with open(p, "rb") as f:
                out[j["key"]] = {**pickle.load(f), "cached": True}
        elif j["key"] not in {t["key"] for t in todo}:
            todo.append(j)
    todo.sort(key=lambda j: -len(j["tr"].frames))

    def done(j, r):
        out[j["key"]] = {**r, "cached": False}
        if cache_dir is not None and "error" not in r:
            _save(_cache_path(cache_dir, j["key"]), r)

    if workers <= 1 or len(todo) <= 1:
        for j in todo:
            done(j, solve_one(j["tr"], ns, camera_ok))
        return out
    saved = {k: os.environ.get(k) for k in ("OMP_NUM_THREADS", "OPENBLAS_NUM_THREADS", "MKL_NUM_THREADS")}
    os.environ.update({k: "1" for k in saved})  # spawned children read these when they import numpy
    try:
        with ProcessPoolExecutor(max_workers=min(workers, len(todo)), mp_context=get_context("spawn")) as ex:
            fut = {ex.submit(solve_one, j["tr"], ns, camera_ok): j for j in todo}
            for f in as_completed(fut):
                done(fut[f], f.result())
    finally:
        for k, v in saved.items():
            if v is None:
                os.environ.pop(k, None)
            else:
                os.environ[k] = v
    return out

    @register_locator("M7", needs=("masks",), scope="segment", classes=(5, 7), aliases="wheel_fit")
def locate_m7(rows: pd.DataFrame, ctx) -> LocatorResult:
    """See the module docstring. Fail reasons: the solve's gate reason, camera_invalid, too_few_frames,
    no_initial_pose, solve_error; kinematic rows of a passed solve are supplied with support "kinematic"."""
    cfg = ctx.config
    n = len(rows)
    res = LocatorResult.empty(n)
    ex = {k: np.full(n, np.nan) for k in ("m7_x", "m7_y", "m7_psi_deg", "m7_n_inliers", "m7_n_outliers",
                                           "m7_sd_along_ft", "m7_sd_across_ft", "m7_sd_heading_deg",
                                           "m7_cov_centre_ee", "m7_cov_centre_en", "m7_cov_centre_nn")}  # fmt: skip
    obj = {k: np.full(n, "", dtype=object) for k in ("m7_support", "m7_near_side", "m7_fail_reason")}
    solve_ok = np.zeros(n, bool)
    per: dict = {}
    res.seg_len_ft, res.seg_wid_ft = np.full(n, np.nan), np.full(n, np.nan)
    res.seg_dims_source = np.full(n, "", dtype=object)
    res.cov = np.full((n, 2, 2), np.nan)
    cam = ctx.camera()
    origin = np.asarray(ctx.origin, float)
    ns = _ns(ctx)
    by_frame = {int(f): g for f, g in ctx.all_boxes.groupby("frame")}
    groups = [np.asarray(v) for v in rows.groupby(["track", "seg"], sort=True).indices.values()]
    jobs, reason = [], {}
    for g in groups:
        sub = rows.iloc[g]
        frames = sub["frame"].to_numpy(int)
        ident = (int(sub["track"].iloc[0]), int(sub["seg"].iloc[0]), int(sub["cls_track"].iloc[0]))
        if cam is None:
            reason[id(g)] = "camera_invalid"
            continue
        if len(frames) > 1 and not (np.diff(frames) == 1).all():
            raise ValueError(f"M7: segment {ident[:2]} does not have consecutive frames")
        if len(frames) < ns["M7_MIN_SUPPORT_FRAMES"]:
            reason[id(g)] = "too_few_frames"
            continue
        m2 = sub[["m2_x", "m2_y"]].to_numpy(float) - origin
        init = init_pose(m2, sub["ref_heading_deg"].to_numpy(float))
        if init is None:
            reason[id(g)] = "no_initial_pose"
            continue
        tr = m7.Track(frames, _obs(sub, ctx, cam, by_frame), cam, origin, init[0], init[1], ident[2], 1.0 / cfg.fps)
        jobs.append({"g": g, "tr": tr, "ident": ident, "key": job_key(tr, ns, True, ident)})
    results = solve_jobs(jobs, ns, ctx.cache_dir, ctx.workers, True)
    solves = []
    for g in groups:
        if id(g) in reason:
            res.fail_reason[g] = reason[id(g)]
            obj["m7_fail_reason"][g] = reason[id(g)]
    for j in jobs:
        g, r = j["g"], results[j["key"]]
        if "error" in r:
            res.fail_reason[g] = "solve_error"
            obj["m7_fail_reason"][g] = "solve_error"
            solves.append({"track": j["ident"][0], "seg": j["ident"][1], "fail_reason": "solve_error",
                           "error": r["error"], "seconds": r.get("seconds", np.nan)})  # fmt: skip
            continue
        sm, fr = r["summary"], r["frames"].reset_index(drop=True)
        fail = sm["fail_reason"]
        passed = fail == ""
        solves.append({"track": j["ident"][0], "seg": j["ident"][1], "cls": j["ident"][2], "n_frames": len(g),
                       "seconds": r.get("seconds", np.nan), "cached": r.get("cached", False),
                       **{k: (json.dumps(v, default=float) if isinstance(v, dict) else v) for k, v in sm.items()}})  # fmt: skip
        pos = fr.set_index("frame").reindex(rows["frame"].to_numpy()[g])
        cen = pos[["e_nat", "n_nat"]].to_numpy(float) + origin
        sup = pos["m7_support"].fillna("").to_numpy(dtype=object)
        cov = pos[["m7_cov_centre_ee", "m7_cov_centre_en", "m7_cov_centre_en", "m7_cov_centre_nn"]].to_numpy(float)
        ex["m7_x"][g], ex["m7_y"][g], ex["m7_psi_deg"][g] = cen[:, 0], cen[:, 1], pos["psi_deg"].to_numpy(float)
        for k in ("m7_n_inliers", "m7_n_outliers", "m7_sd_along_ft", "m7_sd_across_ft", "m7_sd_heading_deg",
                  "m7_cov_centre_ee", "m7_cov_centre_en", "m7_cov_centre_nn"):
            ex[k][g] = pos[k].to_numpy(float)
        obj["m7_support"][g], obj["m7_near_side"][g] = sup, pos["near_side"].astype(object).to_numpy()
        obj["m7_fail_reason"][g] = fail
        solve_ok[g] = passed
        for k, v in _summary_extras(sm).items():
            per.setdefault(k, np.full(n, np.nan if isinstance(v, float) else "", dtype=float if isinstance(v, float) else object))
            per[k][g] = v
        if not passed:
            res.fail_reason[g] = fail
            continue
        res.seg_len_ft[g], res.seg_wid_ft[g] = sm["length_ft"], sm["width_ft"]
        res.seg_dims_source[g] = sm.get("dims_source", "m7_measured")
        native = np.isin(sup, NATIVE_SUPPORT)
        kin = sup == "kinematic"
        ok = (native | kin) & np.isfinite(cen).all(axis=1)
        res.ok[g] = ok
        res.fail_reason[g] = np.where(ok, "", "no_observation")
        res.x[g], res.y[g] = cen[:, 0], cen[:, 1]
        res.orient_deg[g], res.orient_kind[g] = pos["psi_deg"].to_numpy(float), "directed"
        res.support[g] = np.where(kin, "kinematic", "native").astype(object)
        R = cov.reshape(-1, 2, 2) * cfg.kin_meas_scale**2 + (cfg.kin_sd_floor_ft * FT) ** 2 * np.eye(2)
        res.cov[g] = np.where(kin[:, None, None], R, np.nan)
    res.extras.update(ex)
    res.extras.update(obj)
    res.extras.update(per)
    res.extras["m7_solve_ok"] = solve_ok
    ctx.info.setdefault("m7_solves", []).extend(solves)
    return res

    candidates ¶

    Ported from the speed study's m7_candidates.py (sha256 7a664ff6885db247aa0d1b2e942115aece3522ba9e187761d1e7260883ef4261, snapshot docs/study-snapshot-2026-11-10.json). Changes: settings through the per-solve namespace S; image-size and edge defaults resolved at call time; rasterising and the lower outline from tssa.footprint.masks (OpenCV imported on first use).

    M7 wheel-contact candidates from a SAM mask, without a pose (design 2026-11-08-m7, section 5.1).

    Per row: lower outline of the mask inside the bbox -> drop untrustworthy columns (image edge, nearer boxes, leaks below the bbox) -> undistort -> robust one- or two-segment body line (points below it are down-weighted, so tyre bumps do not pull it) -> runs of columns protruding below the line -> runs split at dips -> tyre-size check against the expected tyre diameter D_c at the run's own ground point -> candidate = the run's lowest pixel, with its frozen gate g_c and rejection price kappa_c.

    Geometry inputs: - dh: ground model (image_to_world on DISTORTED pixels, undistort); - cam: camera.Camera (project UTM 3-D -> UNDISTORTED pixels). Contours are lists of (k, 2) distorted pixel polygons (several components allowed).

    def split_contours(arr) -> list[np.ndarray]:
    """(n, 2) array with NaN separator rows -> list of polygons (the m7 cache format)."""
    arr = np.asarray(arr, dtype=float).reshape(-1, 2)
    out, cur = [], []
    for p in arr:
        if np.isnan(p).any():
            if cur:
                out.append(np.array(cur))
            cur = []
        else:
            cur.append(p)
    if cur:
        out.append(np.array(cur))
    return out

    def trusted_mask(outline, box, other_boxes, img_size=None, edge_px=None) -> np.ndarray:
    """Columns of `outline` (distorted px) that are boundary evidence: not at the image edge, not inside a nearer
    box of the same frame (`other_boxes` rows x, y, w, h with a lower bottom edge), not leaking below the bbox."""
    img_size = S.IMG_SIZE if img_size is None else img_size
    edge_px = S.EDGE_PX if edge_px is None else edge_px
    o = np.asarray(outline, dtype=float).reshape(-1, 2)
    W, H = img_size
    x, y, w, h = (float(v) for v in box)
    ok = (o[:, 0] >= edge_px) & (o[:, 0] <= W - 1 - edge_px) & (o[:, 1] <= H - 1 - edge_px)
    ok &= o[:, 1] <= y + h + BELOW_BOTTOM_PX
    for ox, oy, ow, oh in np.asarray(other_boxes, dtype=float).reshape(-1, 4):
        if oy + oh <= y + h:  # not nearer than this vehicle
            continue
        inside = (o[:, 0] >= ox) & (o[:, 0] <= ox + ow) & (o[:, 1] >= oy) & (o[:, 1] <= oy + oh)
        ok &= ~inside
    return ok

    def body_line(uv, thr):
    """Robust one- or two-segment (continuous hinge) body line through undistorted outline points.

    Returns (v_line at each point, info dict: segments, break_u, cost)."""
    u, v = uv[:, 0], uv[:, 1]
    one = np.column_stack([np.ones_like(u), u])
    _, r1, c1 = _irls_line(u, v, one, thr)
    best = (c1, v - r1, {"segments": 1, "break_u": np.nan, "cost": c1})
    n = len(u)
    for q in np.linspace(5, 85, N_BREAKS):
        b = np.percentile(u, q)
        left = (u <= b).sum()
        if min(left, n - left) < max(0.15 * n, 4):
            continue
        two = np.column_stack([np.ones_like(u), u, np.maximum(0.0, u - b)])
        _, r2, c2 = _irls_line(u, v, two, thr)
        if c2 < best[0] and c2 < S.M7_TWO_SEG_GAIN * c1:
            best = (c2, v - r2, {"segments": 2, "break_u": float(b), "cost": c2})
    return best[1], best[2]

    def tyre_px(dh, cam, uv_dist, r_ft) -> np.ndarray:
    """Expected tyre diameter in undistorted px at each distorted pixel's own ground point (no pose)."""
    uv = np.atleast_2d(np.asarray(uv_dist, dtype=float))
    G = np.asarray(dh.image_to_world(uv), dtype=float).reshape(-1, 2)
    z = 2.0 * r_ft * S.FT
    p0 = cam.project(np.column_stack([G, np.zeros(len(G))]))
    p1 = cam.project(np.column_stack([G, np.full(len(G), z)]))
    return np.linalg.norm(p1 - p0, axis=1)

    def extract(contours, box, other_boxes, dh, cam, cls, img_size=None):
    """Candidates of one row and a summary dict. `contours`: list of distorted pixel polygons."""
    summary = {"n_cols": 0, "n_trusted": 0, "segments": 0, "break_u": np.nan, "n_runs": 0, "n_cand": 0,
               "reason": ""}  # fmt: skip
    empty = pd.DataFrame(columns=CAND_COLUMNS)
    if not contours:
        summary["reason"] = "no_mask"
        return empty, summary, None
    mask = rasterise(contours, img_size)
    o = lower_outline(mask, box)
    summary["n_cols"] = len(o)
    ok = trusted_mask(o, box, other_boxes, img_size)
    o = o[ok]
    summary["n_trusted"] = len(o)
    if len(o) < S.M7_MIN_OUTLINE_COLS:
        summary["reason"] = "few_columns"
        return empty, summary, None
    cols = np.round(o[:, 0]).astype(int)
    uv = np.asarray(dh.undistort(o), dtype=float).reshape(-1, 2)
    h = float(box[3])
    thr = S.M7_PROTRUDE_PX * h / 100.0
    v_line, info = body_line(uv, thr)
    summary.update(segments=info["segments"], break_u=info["break_u"])
    depth = uv[:, 1] - v_line
    r_ft = S.M7_WHEEL_R_REF_FT[int(cls)]
    recs = []
    runs = [s for run in _runs(depth > thr, cols) for s in _split_at_dips(run, depth, thr)]
    summary["n_runs"] = len(runs)
    for run in runs:
        bottom = run[uv[run, 1] >= uv[run, 1].max() - 0.75]  # a raster bottom is flat: take its middle
        k = bottom[len(bottom) // 2]
        pu, pv = float(o[bottom, 0].mean()), float(o[k, 1])
        pu_und = float(uv[bottom, 0].mean())
        D = float(tyre_px(dh, cam, np.array([pu, pv]), r_ft)[0])
        width = float(uv[run[-1], 0] - uv[run[0], 0] + 1.0)
        ratio = width / D if D > 0 else np.nan
        ok_w = S.M7_TYRE_W_MIN <= ratio <= S.M7_TYRE_W_MAX
        reject = "" if ok_w else ("narrow" if ratio < S.M7_TYRE_W_MIN else "wide")
        if not reject and depth[k] > S.M7_MAX_DEPTH_TYRES * D:
            ok_w, reject = False, "deep"
        gate = max(S.M7_GATE_MIN_PX, S.M7_GATE_TYRES * D)
        recs.append([len(recs), pu, pv, pu_und, uv[k, 1], float(depth[k]), width, D, ratio, gate,
                     float(rho(gate / S.M7_SIGMA_WHEEL_PX)), bool(ok_w), reject])  # fmt: skip
    cand = pd.DataFrame(recs, columns=CAND_COLUMNS)
    cand = _dedupe(cand)
    summary["n_cand"] = int(cand["accepted"].sum()) if len(cand) else 0
    debug = {"outline": o, "uv": uv, "v_line": v_line, "depth": depth, "thr": thr, "dropped": None}
    return cand, summary, debug

    solver ¶

    Ported from the speed study's m7.py (sha256 d8fc196b1e8375caeaef486eaadcc7314bbf65eaa82c92b26ed67ef8bcbf108f, snapshot docs/study-snapshot-2026-11-10.json). Changes: settings through the per-solve namespace S (an M7Config plus FT, MPS_TO_MPH, IMG_SIZE, EDGE_PX, SIZE_FT) instead of the study's settings module; image-size default resolved at call time; the width-hold decision moved into width_held (same expression) so it can be tested on its own. Speed (2026-11-11, results bit-identical): the sparsity pattern is built with numpy, the matched candidates are packed once per association, and the finite-difference Jacobian is _FDJac (scipy's 2-point sparse scheme with the column groups known in advance) instead of scipy's own, which re-sliced the pattern on every call.

    M7: wheel-anchored rigid-body fit of one bus or truck segment (design 2026-11-08-m7, rev 5, sections 4-8).

    Frames and units: positions are LOCAL metres (the grid's frame; origin converts to UTM for the camera), headings compass radians internally (0 = north, clockwise), f = (sin psi, cos psi), n_left = (-cos psi, sin psi). Image residuals are in UNDISTORTED pixels (cam.project).

    Unknowns: per frame the rear-axle centre R_k (2) and heading psi_k; per solve the statics theta = (L_w, L_f, L_r, W, H[, s_t]) in metres. Observations per frame (FrameObs, frozen before any fit): wheel candidates with frozen rejection prices kappa_c, lower-envelope columns and the two end extents.

    Objective F (section 7.4) = sum over candidates of [matched: rho(|d| / sigma_t) | rejected: kappa_c] + envelope and extent rho terms + 1/2 motion z^2 + 1/2 prior z^2, rho(z) = log(1 + z^2). Association (5.2) is the exact minimiser of F's wheel part for a pose; the pose is fitted by IRLS (a majorise-minimise step for Cauchy) with backtracking on F. The evidence score E (Laplace over the statics) is computed once per completed fit and only compares models (axle configuration, subtype).
